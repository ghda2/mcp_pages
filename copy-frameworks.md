class StoryNavigator {
    constructor() {
        this.currentStep = 1;
        this.init();
    }
    
    init() {
        this.bindEvents();
        this.updateStory();
    }
    
    bindEvents() {
        document.querySelectorAll('.story-step').forEach(button => {
            button.addEventListener('click', (e) => {
                this.currentStep = parseInt(e.currentTarget.dataset.step);
                this.updateStory();
            });
        });
    }
    
    updateStory() {
        // Update step indicators
        document.querySelectorAll('.story-step').forEach((step, index) => {
            const stepNumber = index + 1;
            const button = step.querySelector('div');
            
            if (stepNumber <= this.currentStep) {
                step.classList.add('active');
                button.classList.remove('bg-gray-300', 'text-gray-600');
                button.classList.add('bg-blue-500', 'text-white');
            } else {
                step.classList.remove('active');
                button.classList.remove('bg-blue-500', 'text-white');
                button.classList.add('bg-gray-300', 'text-gray-600');
            }
        });
        
        // Update panels
        document.querySelectorAll('.story-panel').forEach(panel => {
            panel.classList.remove('active');
        });
        
        const activePanel = document.querySelector(`[data-panel="${this.currentStep}"]`);
        if (activePanel) {
            activePanel.classList.add('active');
        }
    }
}

new StoryNavigator();
</script>
